CREATE DATABASE IF NOT EXISTS inclusion_laboral
  DEFAULT CHARACTER SET utf8mb4
  DEFAULT COLLATE utf8mb4_unicode_ci;

USE inclusion_laboral;

-- =========================================================
-- Tablas maestras
-- =========================================================
CREATE TABLE tipo_discapacidad (
  id INT AUTO_INCREMENT PRIMARY KEY,
  discapacidad VARCHAR(100) NOT NULL,
  descripcion TEXT
) ENGINE=InnoDB;

CREATE TABLE provincia (
  id_provincia INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL
) ENGINE=InnoDB;

CREATE TABLE localidad (
  id_localidad INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL,
  id_provincia INT NOT NULL,
  CONSTRAINT fk_localidad_provincia
    FOREIGN KEY (id_provincia) REFERENCES provincia(id_provincia)
    ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB;

-- =========================================================
-- Usuarios base
-- =========================================================
CREATE TABLE usuario (
  id_usuario INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(100),
  apellido VARCHAR(100),
  email VARCHAR(150) UNIQUE,
  clave VARCHAR(255),
  tipo_dni VARCHAR(10),
  dni VARCHAR(20),
  telefono VARCHAR(20),
  tipo ENUM('Persona', 'Empresa', 'Administrador') NOT NULL
) ENGINE=InnoDB;

-- Persona depende fuertemente de usuario → CASCADE
CREATE TABLE persona (
  id_persona INT PRIMARY KEY,
  id_tipodiscapacidad INT NULL,
  cv LONGBLOB,
  certificaciones TEXT,
  CONSTRAINT fk_persona_usuario
    FOREIGN KEY (id_persona) REFERENCES usuario(id_usuario)
    ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT fk_persona_tipodiscapacidad
    FOREIGN KEY (id_tipodiscapacidad) REFERENCES tipo_discapacidad(id)
    ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB;

-- Empresa depende fuertemente de usuario → CASCADE
CREATE TABLE empresa (
  id_empresa INT PRIMARY KEY,
  nombre_empresa VARCHAR(150),
  razon_social VARCHAR(150),
  condicion_social VARCHAR(100),
  documento VARCHAR(50),
  calle VARCHAR(100),
  numero VARCHAR(20),
  codigo_postal VARCHAR(20),
  telefono VARCHAR(50),
  pais VARCHAR(100),
  industria VARCHAR(100),
  cantidad_empleados INT,
  politica_inclusion TEXT,
  datos_contacto TEXT,
  CONSTRAINT fk_empresa_usuario
    FOREIGN KEY (id_empresa) REFERENCES usuario(id_usuario)
    ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB;

-- =========================================================
-- Empleo
-- =========================================================
CREATE TABLE ofertas_laborales (
  id INT AUTO_INCREMENT PRIMARY KEY,
  id_empresa INT NOT NULL,
  titulo VARCHAR(150),
  descripcion TEXT,
  tipo_modalidad VARCHAR(50),
  tipo_trabajo VARCHAR(50),
  carga_horaria INT,
  salario_estimado DECIMAL(10,2),
  id_tipocapacidad INT NULL,
  id_provincia INT NULL,
  id_localidad INT NULL,
  fecha_publicacion DATETIME DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_oferta_empresa
    FOREIGN KEY (id_empresa) REFERENCES empresa(id_empresa)
    ON DELETE CASCADE ON UPDATE CASCADE,
  -- Datos maestros: si llegan a eliminar/editar, no queremos borrar ofertas → SET NULL / RESTRICT
  CONSTRAINT fk_oferta_tipocapacidad
    FOREIGN KEY (id_tipocapacidad) REFERENCES tipo_discapacidad(id)
    ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT fk_oferta_provincia
    FOREIGN KEY (id_provincia) REFERENCES provincia(id_provincia)
    ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT fk_oferta_localidad
    FOREIGN KEY (id_localidad) REFERENCES localidad(id_localidad)
    ON DELETE SET NULL ON UPDATE CASCADE,
  INDEX idx_oferta_empresa (id_empresa),
  INDEX idx_oferta_geo (id_provincia, id_localidad),
  INDEX idx_oferta_tipo (id_tipocapacidad)
) ENGINE=InnoDB;

CREATE TABLE postulaciones (
  id INT AUTO_INCREMENT PRIMARY KEY,
  id_persona INT NOT NULL,
  id_oferta INT NOT NULL,
  fecha_postulacion DATETIME DEFAULT CURRENT_TIMESTAMP,
  estado ENUM('pendiente', 'en revisión', 'aceptado', 'rechazado') NOT NULL,
  CONSTRAINT fk_post_persona
    FOREIGN KEY (id_persona) REFERENCES persona(id_persona)
    ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT fk_post_oferta
    FOREIGN KEY (id_oferta) REFERENCES ofertas_laborales(id)
    ON DELETE CASCADE ON UPDATE CASCADE,
  UNIQUE KEY uk_persona_oferta (id_persona, id_oferta)
) ENGINE=InnoDB;

-- =========================================================
-- Cursos
-- =========================================================
CREATE TABLE curso (
  id_curso INT AUTO_INCREMENT PRIMARY KEY,
  titulo VARCHAR(200) NOT NULL,
  descripcion TEXT,
  duracion_horas INT NOT NULL,
  id_empresa INT NOT NULL,
  publicado TINYINT(1) NOT NULL DEFAULT 1,
  fecha_creacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  fecha_actualizacion DATETIME NULL,
  CONSTRAINT fk_curso_empresa
    FOREIGN KEY (id_empresa) REFERENCES empresa(id_empresa)
    ON DELETE CASCADE ON UPDATE CASCADE,
  INDEX idx_curso_empresa (id_empresa)
) ENGINE=InnoDB;

CREATE TABLE inscripcion (
  id_inscripcion INT AUTO_INCREMENT PRIMARY KEY,
  id_curso INT NOT NULL,
  id_usuario INT NOT NULL,
  fecha_inscripcion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  fecha_inicio DATETIME NULL,
  fecha_completado DATETIME NULL,
  progreso INT DEFAULT 0,
  estado ENUM('inscrito','en_progreso','completado','abandonado') DEFAULT 'inscrito',
  CONSTRAINT fk_insc_curso
    FOREIGN KEY (id_curso) REFERENCES curso(id_curso)
    ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT fk_insc_usuario
    FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
    ON DELETE CASCADE ON UPDATE CASCADE,
  UNIQUE KEY uk_usuario_curso (id_usuario, id_curso),
  INDEX idx_inscripcion_usuario (id_usuario),
  INDEX idx_inscripcion_curso (id_curso)
) ENGINE=InnoDB;

CREATE TABLE certificado (
  id_certificado INT AUTO_INCREMENT PRIMARY KEY,
  codigo_unico VARCHAR(20) UNIQUE NOT NULL,
  id_inscripcion INT NOT NULL,
  id_usuario INT NOT NULL,
  id_curso INT NOT NULL,
  fecha_emision DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  url_validacion VARCHAR(255),
  CONSTRAINT fk_cert_insc
    FOREIGN KEY (id_inscripcion) REFERENCES inscripcion(id_inscripcion)
    ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT fk_cert_usuario
    FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
    ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT fk_cert_curso
    FOREIGN KEY (id_curso) REFERENCES curso(id_curso)
    ON DELETE CASCADE ON UPDATE CASCADE,
  INDEX idx_cert_usuario (id_usuario),
  INDEX idx_cert_codigo (codigo_unico)
) ENGINE=InnoDB;

CREATE TABLE curso_leccion (
  id_leccion INT AUTO_INCREMENT PRIMARY KEY,
  id_curso INT NOT NULL,
  titulo VARCHAR(150) NOT NULL,
  tipo ENUM('pdf','texto') NOT NULL DEFAULT 'pdf',
  url_pdf VARCHAR(255) NULL,
  contenido_texto MEDIUMTEXT NULL,
  orden INT NOT NULL DEFAULT 1,
  creado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_leccion_curso
    FOREIGN KEY (id_curso) REFERENCES curso(id_curso)
    ON DELETE CASCADE ON UPDATE CASCADE,
  INDEX idx_leccion_curso (id_curso),
  INDEX idx_leccion_curso_orden (id_curso, orden)
) ENGINE=InnoDB;

CREATE TABLE curso_leccion_vista (
  id_leccion INT NOT NULL,
  id_usuario INT NOT NULL,
  visto_el DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id_leccion, id_usuario),
  CONSTRAINT fk_leccionvista_leccion
    FOREIGN KEY (id_leccion) REFERENCES curso_leccion(id_leccion)
    ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT fk_leccionvista_usuario
    FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
    ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB;
